**信頼性**: 🔵 *設計文書 architecture.md より*

```rust
if ui.button("Run 2D PDP").clicked()
    && check_params_different(&self.selected_param1, &self.selected_param2) {
    self.pending_compute = Some(Pdp2dComputeRequest {
        param1: self.selected_param1.clone(),
        param2: self.selected_param2.clone(),
        objective: obj_names[self.selected_objective].clone(),
        n_grid: 20,
        model_type: self.selected_model.to_str().to_string(),
    });
}
```
