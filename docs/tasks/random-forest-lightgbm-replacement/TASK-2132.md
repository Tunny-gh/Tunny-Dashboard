### 1. `compute_pdp_2d_lgbm` の実装 🔵

**信頼性**: 🔵 *REQ-101 + dataflow.md 3. 2D PDP データフローより*

**実装ファイル**: `rust_core/src/core/lgbm.rs`

```rust
/// 2D PDP（部分依存プロット）を LightGBM RF で計算する
pub fn compute_pdp_2d_lgbm(
    x_matrix: &[Vec<f64>],
    y: &[f64],
    param1_idx: usize,
    param2_idx: usize,
    n_grid: usize,
) -> Option<(Vec<f64>, Vec<f64>, Vec<Vec<f64>>, f64)> {
    // 2列のみ抽出
    let x2d: Vec<Vec<f64>> = x_matrix.iter()
        .map(|row| vec![row[param1_idx], row[param2_idx]])
        .collect();

    // LightGBM RF 訓練（100木、深度10 — PDP用設定）
    let config = LgbmRfConfig {
        num_iterations: 100,
        max_depth: 10,
        ..Default::default()
    };
    let booster = train_lgbm_rf(&x2d, y, &config)?;

    // グリッド生成
    let min1 = x2d.iter().map(|r| r[0]).fold(f64::INFINITY, f64::min);
    let max1 = x2d.iter().map(|r| r[0]).fold(f64::NEG_INFINITY, f64::max);
    let min2 = x2d.iter().map(|r| r[1]).fold(f64::INFINITY, f64::min);
    let max2 = x2d.iter().map(|r| r[1]).fold(f64::NEG_INFINITY, f64::max);

    let grid1: Vec<f64> = (0..n_grid)
        .map(|i| min1 + (max1 - min1) * i as f64 / (n_grid - 1) as f64)
        .collect();
    let grid2: Vec<f64> = (0..n_grid)
        .map(|i| min2 + (max2 - min2) * i as f64 / (n_grid - 1) as f64)
        .collect();

    // グリッド点で予測
    let grid_points: Vec<Vec<f64>> = grid1.iter()
        .flat_map(|&g1| grid2.iter().map(move |&g2| vec![g1, g2]))
        .collect();
    let flat_z = lgbm_predict(&booster, &grid_points);

    // n_grid x n_grid に変換
    let z_values: Vec<Vec<f64>> = flat_z.chunks(n_grid)
        .map(|c| c.to_vec())
        .collect();

    // R² 計算
    let mse = lgbm_mse(&booster, &x2d, y)?;
    let r_squared = mse_to_r_squared(mse, y);

    Some((grid1, grid2, z_values, r_squared))
}
```

```rust
"random_forest" => {
    let (x_values, y_values, z_values, r_squared) =
        random_forest::compute_pdp_2d_rf(&x_matrix, &y, p1_idx, p2_idx, n_grid)?;
    Some(PdpResult2d { x_values, y_values, z_values, r_squared, uncertainties: None })

```

```rust
"random_forest" => {
    let (x_values, y_values, z_values, r_squared) =
        crate::core::lgbm::compute_pdp_2d_lgbm(
            &x_matrix, &y, p1_idx, p2_idx, n_grid
        )?;
    Some(PdpResult2d { x_values, y_values, z_values, r_squared, uncertainties: None })
}
```
