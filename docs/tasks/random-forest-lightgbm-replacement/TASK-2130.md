/// LightGBM RandomForest モード用ハイパーパラメータ設定
pub struct LgbmRfConfig {
    pub num_iterations: usize,   // 木の本数 (旧 n_trees)
    pub max_depth: i32,          // 木の最大深度 (旧 max_depth)
    pub min_data_in_leaf: i32,   // リーフの最小サンプル数 (旧 min_samples_leaf)
    pub bagging_fraction: f64,   // バギング率 (RF 有効化に必須、< 1.0)
    pub feature_fraction: f64,   // 特徴量サンプリング率
    pub seed: i32,               // 乱数シード
}

impl Default for LgbmRfConfig {
    /// SHAP/MDI 共通デフォルト (64木, 深度10)
    fn default() -> Self {
        Self {
            num_iterations: 64,
            max_depth: 10,
            min_data_in_leaf: 2,
            bagging_fraction: 0.8,
            feature_fraction: 0.8,
            seed: 42,
        }
    }
}
```

/// &[Vec<f64>] と &[f64] から lightgbm::Dataset を作成する
pub fn to_lgbm_dataset(
    x: &[Vec<f64>],
    y: &[f64],
) -> Result<lightgbm::Dataset, lightgbm::Error> {
    lightgbm::Dataset::from_mat(x, y)
}

**信頼性**: 🔵 *architecture.md LightGBM RF パラメータ表 + REQ-001 より*

/// LightGBM RF モデルを訓練する
pub fn train_lgbm_rf(
    x: &[Vec<f64>],
    y: &[f64],
    config: &LgbmRfConfig,
) -> Option<lightgbm::Booster> {
    let dataset = to_lgbm_dataset(x, y).ok()?;
    let params = [
        ("boosting_type", "rf"),
        ("num_iterations", &config.num_iterations.to_string()),
        ("max_depth", &config.max_depth.to_string()),
        ("min_data_in_leaf", &config.min_data_in_leaf.to_string()),
        ("bagging_fraction", &config.bagging_fraction.to_string()),
        ("bagging_freq", "1"),
        ("feature_fraction", &config.feature_fraction.to_string()),
        ("verbose", "-1"),
        ("num_threads", "1"),
        ("seed", &config.seed.to_string()),
        ("objective", "regression"),
    ];
    lightgbm::Booster::train(dataset, &params).ok()
}
```

```rust
/// LightGBM モデルの MSE を計算し、R² も算出する
pub fn lgbm_mse(
    booster: &lightgbm::Booster,
    x_eval: &[Vec<f64>],
    y_eval: &[f64],
) -> Option<f64> {
    if y_eval.is_empty() {
        return None;
    }
    let preds = lgbm_predict(booster, x_eval);
    let mse = preds.iter().zip(y_eval.iter())
        .map(|(p, y)| (p - y).powi(2))
        .sum::<f64>() / y_eval.len() as f64;
    Some(mse)
}

/// MSE から R² を計算するヘルパー
pub fn mse_to_r_squared(mse: f64, y_eval: &[f64]) -> f64 {
    let y_mean = y_eval.iter().sum::<f64>() / y_eval.len() as f64;
    let ss_tot = y_eval.iter().map(|y| (y - y_mean).powi(2)).sum::<f64>();
    if ss_tot == 0.0 { return 0.0; }
    1.0 - (mse * y_eval.len() as f64) / ss_tot
}
```

## 注意事項

- `lightgbm-rs` クレートの API は バージョンによって異なる。`Booster::train`、`Dataset::from_mat` のシグネチャをクレートのドキュメントで確認すること
- `bagging_freq = 1` は RF モード有効化に必須（`bagging_fraction < 1.0` だけでは不足）
- `num_threads = 1` でシングルスレッド動作にする（並列化は将来の NFR として保留）
- lightgbm の `params` は `&[(&str, &str)]` 形式のため、数値は文字列に変換する必要がある
