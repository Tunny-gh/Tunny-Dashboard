### sensitivity_bench.rs 🔵

**ファイル**: `rust_core/benches/sensitivity_bench.rs`

```rust
/// Criterion benchmarks for sensitivity analysis (run_tree_metric_for_all_objectives).
///
/// Run with: cargo bench -p tunny-core --bench sensitivity_bench
use criterion::{criterion_group, criterion_main, BenchmarkId, Criterion};
use std::collections::HashMap;
use tunny_core::dataframe::{select_study, store_dataframes, DataFrame, TrialRow};
use tunny_core::sensitivity::compute_sensitivity_from_df;

fn setup_sensitivity_df(n: usize, n_params: usize, n_objectives: usize) {
    let param_names: Vec<String> = (0..n_params).map(|i| format!("x{}", i)).collect();
    let obj_names: Vec<String> = (0..n_objectives).map(|i| format!("obj{}", i)).collect();
    let rows: Vec<TrialRow> = (0..n)
        .map(|i| {
            let fi = i as f64;
            TrialRow {
                trial_id: i as u32,
                param_display: param_names
                    .iter()
                    .enumerate()
                    .map(|(j, name)| (name.clone(), fi * (j + 1) as f64 * 0.01))
                    .collect::<HashMap<_, _>>(),
                param_category_label: HashMap::new(),
                objective_values: (0..n_objectives).map(|k| fi + k as f64).collect(),
                user_attrs_numeric: HashMap::new(),
                user_attrs_string: HashMap::new(),
                constraint_values: vec![],
            }
        })
        .collect();
    let df = DataFrame::from_trials(&rows, &param_names, &obj_names, &[], &[], 0);
    store_dataframes(vec![df]);
    select_study(0).expect("select_study failed");
}

fn bench_sensitivity(c: &mut Criterion) {
    let mut group = c.benchmark_group("sensitivity_objectives");
    for &n_obj in &[1usize, 2, 4, 8] {
        group.bench_with_input(
            BenchmarkId::new("rf_anova", n_obj),
            &n_obj,
            |b, &n_obj| {
                setup_sensitivity_df(200, 5, n_obj);
                b.iter(|| {
                    tunny_core::dataframe::with_active_df(|df| {
                        compute_sensitivity_from_df(df)
                    })
                });
            },
        );
    }
    group.finish();
}

criterion_group!(benches, bench_sensitivity);
criterion_main!(benches);
```

### rf_bench.rs 🔵

**ファイル**: `rust_core/benches/rf_bench.rs`

```rust
/// Criterion benchmarks for RandomForest::train.
///
/// Run with: cargo bench -p tunny-core --bench rf_bench
use criterion::{criterion_group, criterion_main, BenchmarkId, Criterion};
use tunny_core::core::random_forest::forest::RandomForest;  // pub(crate) の場合は調整要

fn make_data(n: usize, p: usize) -> (Vec<Vec<f64>>, Vec<f64>) {
    let x: Vec<Vec<f64>> = (0..n)
        .map(|i| (0..p).map(|j| (i * p + j) as f64 * 0.01).collect())
        .collect();
    let y: Vec<f64> = (0..n).map(|i| i as f64 * 0.1).collect();
    (x, y)
}

fn bench_rf_train(c: &mut Criterion) {
    let mut group = c.benchmark_group("rf_train_n_trees");
    for &n_trees in &[10usize, 50, 100] {
        group.bench_with_input(
            BenchmarkId::new("n_trees", n_trees),
            &n_trees,
            |b, &n_trees| {
                let (x, y) = make_data(200, 5);
                b.iter(|| RandomForest::train(&x, &y, n_trees, 5, 2, 42));
            },
        );
    }
    group.finish();
}

criterion_group!(benches, bench_rf_train);
criterion_main!(benches);
```

> `RandomForest::train` が `pub(crate)` の場合、ベンチマーク用に `pub` に変更するか、
> `tunny_core` クレートに re-export する薄いラッパーを追加する。
> 既存の公開状況をコンパイル時に確認して対応する。

///
/// Run with: cargo bench -p tunny-core --bench permutation_bench
use criterion::{criterion_group, criterion_main, BenchmarkId, Criterion};

    for &p in &[3usize, 10, 20] {
        group.bench_with_input(BenchmarkId::new("p", p), &p, |b, &p| {

            let x: Vec<Vec<f64>> = (0..n)
                .map(|i| (0..p).map(|j| (i * p + j) as f64 * 0.01).collect())
                .collect();
            let y: Vec<f64> = (0..n).map(|i| i as f64 * 0.1).collect();
            b.iter(|| compute_permutation_importances(&x, &y));
        });
    }
    group.finish();
}

criterion_group!(benches, bench_permutation);
criterion_main!(benches);
