### show() メソッド実装 🔵

**信頼性**: 🔵 *architecture.md「ParallelCoordinates」・dataflow.md「ParallelCoordinates データフロー」より*

`egui-app/src/ui/widgets/parallel_coords.rs` に追加:

```rust
use crate::state::app_state::TrialRow;

impl ParallelCoordsChart {
    pub fn show(
        &mut self,
        ui: &mut egui::Ui,
        trial_rows: &[TrialRow],
        param_names: &[String],
        obj_names: &[String],
    ) {
        if trial_rows.is_empty() {
            ui.centered_and_justified(|ui| {
                ui.label(egui::RichText::new("No trial data.").weak());
            });
            return;
        }

        // 全変数名を統合（パラメータ + 目的関数）
        let all_names: Vec<&str> = param_names.iter()
            .chain(obj_names.iter())
            .map(|s| s.as_str())
            .collect();
        let n_axes = all_names.len();
        if n_axes < 2 { return; }

        let available = ui.available_rect_before_wrap();
        let painter = ui.painter();

        // 軸の X 座標を均等配置
        let axis_margin = 40.0;
        let axis_x: Vec<f32> = (0..n_axes)
            .map(|i| available.min.x + axis_margin + (available.width() - 2.0 * axis_margin) * i as f32 / (n_axes - 1) as f32)
            .collect();
        let axis_top = available.min.y + 30.0;
        let axis_bottom = available.max.y - 10.0;

        // 軸ラベルと縦線を描画
        for (i, &name) in all_names.iter().enumerate() {
            let x = axis_x[i];
            painter.line_segment(
                [egui::pos2(x, axis_top), egui::pos2(x, axis_bottom)],
                egui::Stroke::new(1.5, egui::Color32::from_gray(150)),
            );
            painter.text(
                egui::pos2(x, available.min.y + 15.0),
                egui::Align2::CENTER_CENTER,
                name,
                egui::FontId::proportional(10.0),
                ui.visuals().text_color(),
            );
        }

        // 各軸のデータ最小値・最大値を計算して正規化して折れ線描画
        for trial in trial_rows {
            let mut points: Vec<egui::Pos2> = Vec::new();
            let mut valid = true;
            for (i, &name) in all_names.iter().enumerate() {
                // TrialRow から変数値を取得（param_values または obj_values）
                let val_opt = trial.param_values.get(name).or_else(|| trial.obj_values.get(name));
                let Some(&val) = val_opt else { valid = false; break; };

                // 全試行での min/max を取得して正規化
                let (min_v, max_v) = trial_rows.iter()
                    .filter_map(|t| t.param_values.get(name).or_else(|| t.obj_values.get(name)))
                    .fold((f64::INFINITY, f64::NEG_INFINITY), |(mn, mx), &v| (mn.min(v), mx.max(v)));

                let norm = self.normalize_value(val, min_v, max_v);
                let y = self.normalized_to_screen_y(norm, axis_top, axis_bottom);
                points.push(egui::pos2(axis_x[i], y));
            }
            if valid && points.len() >= 2 {
                for pair in points.windows(2) {
                    painter.line_segment(
                        [pair[0], pair[1]],
                        egui::Stroke::new(0.8, egui::Color32::from_rgba_unmultiplied(100, 150, 220, 80)),
                    );
                }
            }
        }

        ui.allocate_rect(available, egui::Sense::hover());
    }
}
```
