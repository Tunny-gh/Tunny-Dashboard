### show() メソッド実装 🔵

**信頼性**: 🔵 *architecture.md「ScatterMatrix」・dataflow.md「ScatterMatrix データフロー」より*

`egui-app/src/ui/widgets/scatter_matrix.rs` に追加:

```rust
use crate::state::app_state::TrialRow;

impl ScatterMatrix {
    pub fn show(
        &mut self,
        ui: &mut egui::Ui,
        trial_rows: &[TrialRow],
        param_names: &[String],
        obj_names: &[String],
    ) {
        if trial_rows.is_empty() {
            ui.centered_and_justified(|ui| {
                ui.label(egui::RichText::new("No trial data.").weak());
            });
            return;
        }

        // 全変数名リスト（パラメータ + 目的関数）
        let all_names: Vec<&str> = param_names.iter()
            .chain(obj_names.iter())
            .map(|s| s.as_str())
            .collect();
        let n = all_names.len();
        if n == 0 { return; }

        let available = ui.available_rect_before_wrap();
        let cell_w = available.width() / n as f32;
        let cell_h = available.height() / n as f32;

        for row in 0..n {
            for col in 0..n {
                let min = available.min + egui::vec2(col as f32 * cell_w, row as f32 * cell_h);
                let cell_rect = egui::Rect::from_min_size(min, egui::vec2(cell_w, cell_h));
                let mut child_ui = ui.new_child(
                    egui::UiBuilder::new()
                        .max_rect(cell_rect)
                        .layout(egui::Layout::top_down(egui::Align::LEFT)),
                );

                if row == col {
                    self.draw_histogram_cell(&mut child_ui, trial_rows, all_names[row], cell_rect);
                } else if col > row {
                    self.draw_correlation_cell(&mut child_ui, trial_rows, all_names[row], all_names[col], cell_rect);
                } else {
                    self.draw_scatter_cell(&mut child_ui, trial_rows, all_names[col], all_names[row], cell_rect);
                }
            }
        }

        ui.allocate_rect(available, egui::Sense::hover());
    }
}
```
