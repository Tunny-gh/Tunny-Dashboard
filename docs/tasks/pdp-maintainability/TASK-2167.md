// 変更前（~8行のブロック）
let ss_tot: f64 = y.iter().map(|&v| (v - y_mean).powi(2)).sum();
let ss_res: f64 = x_norm.iter().zip(y.iter()).map(|(xi, &yi)| {
    let pred = gaussian_process::predict_mean(&model, xi) * y_std + y_mean;
    (yi - pred).powi(2)
}).sum();
let r_squared = if ss_tot < f64::EPSILON { 1.0 } else { 1.0 - ss_res / ss_tot };

// 変更後
let y_pred_2d: Vec<f64> = x_norm.iter()
    .map(|xi| gaussian_process::predict_mean(&model, xi) * y_std + y_mean)
    .collect();
let r_sq = r_squared(y, &y_pred_2d);
