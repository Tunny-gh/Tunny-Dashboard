### 2. extractData 関数の実装 🔵

**信頼性**: 🔵 *dataflow.md データ転送詳細（REQ-001-02）より*

**実装ファイル**: `frontend/src/stores/analysisStore.ts`

```typescript
function extractData(
  param1: string,
  param2: string,
  objective: string,
  df: DataFrameInfo
): { xFlat: Float64Array; y: Float64Array; nSamples: number; p1Idx: number; p2Idx: number } {
  const n = df.nRows;
  const xFlat = new Float64Array(n * 2);
  // xFlat = [param1_0..param1_n, param2_0..param2_n] (列優先)
  const p1Col = df.columns[param1];
  const p2Col = df.columns[param2];
  for (let i = 0; i < n; i++) {
    xFlat[i] = p1Col[i];
    xFlat[n + i] = p2Col[i];
  }
  const y = new Float64Array(df.columns[objective]);
  return { xFlat, y, nSamples: n, p1Idx: 0, p2Idx: 1 };
}
```

```typescript
// kriging/sparse_kriging → Worker 経由
if (modelType === 'kriging' || modelType === 'sparse_kriging') {
  const { xFlat, y, nSamples, p1Idx, p2Idx } = extractData(param1, param2, objective, df);
  const cacheKey = `${modelType}_${param1}_${param2}_${objective}_${nGrid}`;
  
  worker.postMessage(
    { type: 'compute', xFlat, y, nSamples, param1Idx: p1Idx, param2Idx: p2Idx, nGrid, modelType, cacheKey },
    [xFlat.buffer, y.buffer]  // Transferable: ゼロコピー
  );
  set({ isComputingSurface: true });
} else {
  // ridge/random_forest: 既存の同期 WASM 処理
  // ...
}
```

### 4. Worker onmessage ハンドラ 🔵

**信頼性**: 🔵 *dataflow.md Phase 2 シーケンス図・REQ-001-04 より*

```typescript
worker.onmessage = (e: MessageEvent<KrigingWorkerOutput>) => {
  if (e.data.type === 'result') {
    const { result, cacheKey } = e.data;
    set((state) => ({
      surface3dCache: new Map(state.surface3dCache).set(cacheKey!, result),
      isComputingSurface: false,
    }));
  } else if (e.data.type === 'error') {
    set({ surface3dError: e.data.error, isComputingSurface: false });
  }
};
```
