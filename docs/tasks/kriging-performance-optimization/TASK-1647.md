## タスク概要

TASK-1646 で作成した Worker スケルトンに完全な実装を追加する。Worker は以下のメッセージを処理する:
- `type: 'init'`: WASM バイナリ（base64）を受け取り、独立した WASM インスタンスを初期化。完了後 `{ type: 'ready' }` を返す。
- `type: 'compute'`: `xFlat`, `y` などのデータを受け取り `computeKrigingRaw` を実行。結果を `{ type: 'result', result, cacheKey }` で返す。

export const KRIGING_WORKER_SRC = `
  let wasmModule = null;
  let computeKrigingRaw = null;

  self.onmessage = async function(e) {
    const { type, wasmBase64, xFlat, y, nSamples, param1Idx, param2Idx, nGrid, modelType, cacheKey } = e.data;

    if (type === 'init') {
      try {
        // base64 → Uint8Array → WebAssembly.Module
        const wasmBinary = base64ToUint8Array(wasmBase64);
        const wasmCompiledModule = await WebAssembly.compile(wasmBinary);
        // wasm-bindgen 生成の init 関数で初期化
        const wasm = await initWasm({ module: wasmCompiledModule });
        computeKrigingRaw = wasm.computeKrigingRaw;
        self.postMessage({ type: 'ready' });
      } catch (err) {
        self.postMessage({ type: 'error', error: err.toString() });
      }
      return;
    }

    if (type === 'compute') {
      try {
        const result = computeKrigingRaw(xFlat, y, nSamples, param1Idx, param2Idx, nGrid, modelType);
        self.postMessage({ type: 'result', result, cacheKey });
      } catch (err) {
        self.postMessage({ type: 'error', error: err.toString(), cacheKey });
      }
      return;
    }
  };

  function base64ToUint8Array(base64) {
    const binary = atob(base64);
    const bytes = new Uint8Array(binary.length);

    }
    return bytes;
