```rust
fn show_study_info(ui: &mut egui::Ui, app_state: &AppState) {
    if let Some(ctx) = &app_state.current_study {
        ui.label(format!("Study: {}", ctx.meta.study_name));
        ui.label(format!("試行数: {}", ctx.meta.n_trials));
        ui.label(format!("目的数: {}", ctx.meta.n_objectives));
        ui.label(format!("変数数: {}", ctx.meta.n_params));
        ui.label(format!(
            "選択: {} / {}",
            app_state.selected_indices.len(),
            ctx.trial_rows.len()
        ));
    }
}
```

### 3. チャート表示選択リスト 🔵

**信頼性**: 🔵 *設計文書より*

CheckBox で各 ChartId のオン/オフを切り替える。

```rust
fn show_chart_selection(ui: &mut egui::Ui, layout: &mut LayoutState) {
    ui.collapsing("チャート表示", |ui| {
        for chart_id in ChartId::all() {
            let mut visible = layout.visible_charts.contains(&chart_id);
            if ui.checkbox(&mut visible, chart_id.label()).changed() {
                layout.toggle_chart(chart_id);
            }
        }
    });
}
```

### 4. 変数フィルタースライダー 🔵

**信頼性**: 🔵 *設計文書より*

各 param_name に対して RangeSlider を表示する。最小・最大値は TrialRow の実データから算出する。スライダー変更時に `app_state.set_filter(axis, min, max)` を呼び出す。

```rust
fn show_filter_sliders(ui: &mut egui::Ui, app_state: &mut AppState) {
    if let Some(ctx) = &app_state.current_study {
        ui.collapsing("フィルター", |ui| {
            for param_name in ctx.param_names() {
                let (data_min, data_max) = ctx.param_range(&param_name);
                let (mut filter_min, mut filter_max) = app_state
                    .filter_ranges
                    .get(&param_name)
                    .copied()
                    .unwrap_or((data_min, data_max));
                // RangeSlider の描画と変更検知
            }
        });
    }
}
```

```rust
fn show_color_mode(ui: &mut egui::Ui, app_state: &mut AppState) {
    ui.label("カラーモード:");
    egui::ComboBox::from_label("")
        .selected_text(app_state.color_mode.label())
        .show_ui(ui, |ui| {
            ui.selectable_value(&mut app_state.color_mode, ColorMode::ParetoRank, "Pareto ランク");
            ui.selectable_value(&mut app_state.color_mode, ColorMode::TrialNumber, "試行番号");
            // ObjectiveValue と ClusterId は動的に選択肢を生成
        });
}
```
