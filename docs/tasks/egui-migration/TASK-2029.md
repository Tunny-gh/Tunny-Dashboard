    pub capacity: usize,

    pub fn new(device: &wgpu::Device, capacity: usize) -> Self {
        let positions = device.create_buffer(&wgpu::BufferDescriptor {
            label: Some("positions"),

### 4. カスタム WGSL シェーダー (render/scatter.wgsl) 🔵

**信頼性**: 🔵 *設計文書より*

`egui-app/src/render/scatter.wgsl` を作成する。点を四角形スプライト（billboard）として描画するシェーダー。

```wgsl
struct ViewProjection {
    matrix: mat4x4<f32>,
}

@group(0) @binding(0) var<uniform> vp: ViewProjection;

struct VertexInput {
    @location(0) position: vec4<f32>,
    @location(1) color: vec4<f32>,
    @location(2) size: f32,
}

struct VertexOutput {
    @builtin(position) clip_position: vec4<f32>,
    @location(0) color: vec4<f32>,
    @location(1) point_coord: vec2<f32>,
}

@vertex
fn vs_main(vertex: VertexInput, @builtin(vertex_index) v_idx: u32) -> VertexOutput {
    // billboard sprite 展開ロジック
}

@fragment
fn fs_main(in: VertexOutput) -> @location(0) vec4<f32> {
    // 円形クリッピング
    let dist = length(in.point_coord);
    if dist > 1.0 { discard; }
    return in.color;
}
```
