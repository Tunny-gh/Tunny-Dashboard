### 2. バックグラウンドで select_study 実行 🔵

**信頼性**: 🔵 *設計文書より*

```rust
fn select_study(&mut self, study_meta: StudyMeta) {
    self.app_state.clear();
    self.app_state.is_loading = true;
    let tx = self.tx.clone();
    let meta = study_meta.clone();
    spawn_task(tx, move || {
        match tunny_core::io::journal::select_study(meta.study_id) {
            Ok(result) => {
                let pareto = tunny_core::multi_objective::pareto::compute_pareto_ranks(&result)
                    .unwrap_or_default();
                let gpu_data = build_gpu_buffer_data(&result, &pareto);
                AppMessage::StudySelected {
                    meta,
                    trial_rows: result,
                    gpu_data,
                    pareto_indices: pareto,
                }
            }
            Err(e) => AppMessage::Error(e.to_string()),
        }
    });
}
```
