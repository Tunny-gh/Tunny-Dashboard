## 実装詳細

### 1. extract_axis_values()関数 🔵

**信頼性**: 🔵 *dataflow.md Section 2 Compute Layer フローより*

```rust
pub fn extract_axis_values(
    axis_id: &str,
    mcdm_result: &McdmResult,
    trials: &[TrialRow],
) -> Result<Vec<f64>, ScatterError> {
    // 1. 目的関数か MCDM スコアかを判定
    if axis_id.starts_with("Objective") {
        // 目的関数値を抽出
        extract_objective_values(axis_id, trials)
    } else {
        // MCDM スコア値を抽出
        extract_mcdm_score_values(axis_id, mcdm_result)
    }
}

fn extract_objective_values(
    axis_id: &str,
    trials: &[TrialRow],
) -> Result<Vec<f64>, ScatterError> {
    // "Objective0" → index 0
    let index_str = axis_id.strip_prefix("Objective")
        .ok_or(ScatterError::AxisNotFound { 
            axis_id: axis_id.to_string() 
        })?;
    let index: usize = index_str.parse()
        .map_err(|_| ScatterError::InvalidAxisData {
            axis_id: axis_id.to_string(),
            reason: "Invalid objective index".to_string(),
        })?;

    let values: Vec<f64> = trials.iter()
        .filter_map(|trial| {
            trial.objectives.get(index)
                .copied()
                .filter(|v| v.is_finite())
        })
        .collect();

    if values.is_empty() {
        Err(ScatterError::InvalidAxisData {
            axis_id: axis_id.to_string(),
            reason: "No valid data for this objective".to_string(),
        })
    } else {
        Ok(values)
    }
}

fn extract_mcdm_score_values(
    axis_id: &str,
    mcdm_result: &McdmResult,
) -> Result<Vec<f64>, ScatterError> {
    match mcdm_result {
        McdmResult::Vikor(vikor) => {
            match axis_id {
                "VIKOR_Q" => Ok(vikor.q_values.clone()),
                "VIKOR_S" => Ok(vikor.s_values.clone()),
                "VIKOR_R" => Ok(vikor.r_values.clone()),
                _ => Err(ScatterError::AxisNotFound {
                    axis_id: axis_id.to_string(),
                }),
            }
        }
        McdmResult::Topsis(topsis) => {
            match axis_id {
                "TOPSIS_Score" => Ok(topsis.scores.clone()),
                _ => Err(ScatterError::AxisNotFound {
                    axis_id: axis_id.to_string(),
                }),
            }
        }
        McdmResult::PrometheeI(promethee) | 
        McdmResult::PrometheeII(promethee) => {
            match axis_id {
                "Phi+" => Ok(promethee.phi_positive.clone()),
                "Phi-" => Ok(promethee.phi_negative.clone()),
                _ => Err(ScatterError::AxisNotFound {
                    axis_id: axis_id.to_string(),
                }),
            }
        }
    }
}
```

#[test]
fn test_extract_objective_values() {
    let trials = vec![
        TrialRow { objectives: vec![100.0, 50.0], ... },
        TrialRow { objectives: vec![150.0, 60.0], ... },
    ];
    
    let values = extract_axis_values(
        "Objective0",
        &mock_mcdm_result(),
        &trials
    ).unwrap();
    
    assert_eq!(values, vec![100.0, 150.0]);
}
