```rust
impl McdmScatterChart {
    pub fn show(&mut self, ui: &mut egui::Ui, app_state: &mut AppState) {
        // 既存のUI制御行（TASK-1502で実装）
        // self.render_controls(ui, app_state);
        
        // プロット描画
        self.render_plot(ui, app_state);
    }
    
    fn render_plot(&mut self, ui: &mut egui::Ui, app_state: &mut AppState) {
        let plot = egui_plot::Plot::new("mcdm_scatter")
            .legend(egui_plot::Legend::default())
            .allow_drag(true)
            .allow_scroll(true)
            .allow_zoom(true)
            .x_axis_label(&self.x_axis)
            .y_axis_label(&self.y_axis);
        
        plot.show(ui, |plot_ui| {
            self.render_points(plot_ui);
        });
    }
}
```

### 2. ポイント描画 🔵

**信頼性**: 🔵 *architecture.md Section 6 egui_plot 描画より*

```rust
fn render_points(&self, plot_ui: &mut egui_plot::PlotUi) {
    if let Some(points) = &self.display_rows_cache {
        // Group points by color for efficient rendering
        let mut red_points = Vec::new();
        let mut orange_points = Vec::new();
        let mut yellow_points = Vec::new();
        let mut gray_points = Vec::new();
        
        for (x, y, color) in points {
            let point = [*x, *y];
            match color {
                Color32::from_rgb(255, 0, 0) => red_points.push(point),
                Color32::from_rgb(255, 165, 0) => orange_points.push(point),
                Color32::from_rgb(255, 255, 0) => yellow_points.push(point),
                Color32::from_rgb(200, 200, 200) => gray_points.push(point),
                _ => gray_points.push(point),  // Default
            }
        }
        
        // Render each color group (render order matters)
        if !gray_points.is_empty() {
            plot_ui.points(
                egui_plot::Points::new(gray_points)
                    .color(Color32::from_rgb(200, 200, 200))
                    .radius(3.0)
            );
        }
        
        if !yellow_points.is_empty() {
            plot_ui.points(
                egui_plot::Points::new(yellow_points)
                    .color(Color32::from_rgb(255, 255, 0))
                    .radius(4.0)
            );
        }
        
        if !orange_points.is_empty() {
            plot_ui.points(
                egui_plot::Points::new(orange_points)
                    .color(Color32::from_rgb(255, 165, 0))
                    .radius(4.0)
            );
        }
        
        if !red_points.is_empty() {
            plot_ui.points(
                egui_plot::Points::new(red_points)
                    .color(Color32::from_rgb(255, 0, 0))
                    .radius(5.0)  // Slightly larger for visibility
            );
        }
    }
}
```

```rust
#[test]
fn test_point_grouping() {
    let points = vec![
        (0.5, 0.5, Color32::from_rgb(255, 0, 0)),
        (0.6, 0.6, Color32::from_rgb(200, 200, 200)),
    ];
    
    let mut red_count = 0;
    let mut gray_count = 0;
    
    for (_, _, color) in &points {
        match color {
            _ if color == &Color32::from_rgb(255, 0, 0) => red_count += 1,
            _ if color == &Color32::from_rgb(200, 200, 200) => gray_count += 1,
            _ => {}
        }
    }
    
    assert_eq!(red_count, 1);
    assert_eq!(gray_count, 1);
}
```
