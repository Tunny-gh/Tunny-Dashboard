## 単体テスト例

```rust
#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_full_workflow_scatter_plot() {
        // 1. MCDM結果取得
        let mcdm_result = create_mock_vikor_result();
        
        // 2. 軸オプション生成
        let options = get_axis_options(&mcdm_result, &["Height", "Weight"]);
        assert!(options.len() >= 5);  // Objective + VIKOR scores
        
        // 3. 軸値抽出
        let x_values = extract_axis_values("Objective0", &mcdm_result, &mock_trials()).unwrap();
        assert_eq!(x_values.len(), 100);
        
        // 4. 正規化
        let x_normalized = normalize_values(&x_values);
        assert!(x_normalized.iter().all(|v| (0.0..=1.0).contains(v)));
        
        // 5. ダウンサンプリング
        let downsampled = downsample_points(&mock_points_300(), 300);
        assert!(downsampled.len() <= 300);
        
        // 6. 色分けマッピング
        let colors = compute_point_colors(&mcdm_result.ranked_indices, TopN::Top5);
        assert_eq!(colors.len(), mcdm_result.ranked_indices.len());
        
        // 7. キャッシング
        assert!(downsampled.len() > 0);
    }
}
```

## 統合テスト例

```rust
#[cfg(test)]
mod integration_tests {
    use super::*;

    #[test]
    fn test_end_to_end_scatter_plot_rendering() {
        // e2e: UI制御 → 計算 → キャッシュ → 描画
        let mut app_state = create_mock_app_state();
        let mut chart = McdmScatterChart::new();
        
        // ユーザーが X軸を変更
        chart.x_axis = "VIKOR_Q".to_string();
        chart.update_cache_key(app_state.trials.len());
        
        // キャッシュが無効化
        assert!(chart.display_rows_cache.is_none());
        assert!(chart.pending_compute.is_some());
        
        // 計算実行
        let result = compute_scatter_points(&app_state, &chart);
        
        // メッセージ処理
        let message = AppMessage::McdmScatterComputed {
            points: result.points,
            metadata: result.metadata,
        };
        
        // キャッシュ更新
        if let AppMessage::McdmScatterComputed { points, .. } = message {
            chart.display_rows_cache = Some(points);
        }
        
        // 描画準備完了
        assert!(chart.display_rows_cache.is_some());
    }
}
```
