```rust
/// 重みベクターの合計が 1.0 になるよう正規化する。
/// 合計が 0 に近い場合は均等分割する。
pub fn normalize_weights(weights: &mut [f64]) {
    let sum: f64 = weights.iter().sum();
    if sum > f64::EPSILON {
        for w in weights.iter_mut() {
            *w /= sum;
        }
    } else if !weights.is_empty() {
        let uniform = 1.0 / weights.len() as f64;
        for w in weights.iter_mut() {
            *w = uniform;
        }
    }
}
```

```rust
#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn normalize_weights_sum_to_one() {
        let mut weights = vec![1.0, 2.0, 1.0];
        normalize_weights(&mut weights);
        let sum: f64 = weights.iter().sum();
        assert!((sum - 1.0).abs() < 1e-10);
    }

    #[test]
    fn normalize_weights_zero_sum() {
        let mut weights = vec![0.0, 0.0, 0.0];
        normalize_weights(&mut weights);
        let expected = 1.0 / 3.0;
        for &w in &weights {
            assert!((w - expected).abs() < 1e-10);
        }
    }
}
```
