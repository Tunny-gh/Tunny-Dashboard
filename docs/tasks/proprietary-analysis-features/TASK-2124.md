```rust
use egui::{Ui, Color32};
use crate::state::app_state::AppState;

/// ビュー切り替えタブの状態
#[derive(PartialEq, Clone, Copy)]
pub enum ComparisonView {
    Stats,
    HvHistory,
    ParetoFront,
    KdeDistribution,
}

impl Default for ComparisonView {
    fn default() -> Self { Self::Stats }
}

/// マルチスタディ比較パネルを表示する
pub fn show_comparison_panel(
    ui: &mut Ui,
    app_state: &AppState,
    active_view: &mut ComparisonView,
) {
    if app_state.comparison_studies.is_empty() {
        ui.centered_and_justified(|ui| {
            ui.label("比較スタディを追加してください（ツールバーの「＋」ボタン）");
        });
        return;
    }

    // タブ切り替え
    ui.horizontal(|ui| {
        ui.selectable_value(active_view, ComparisonView::Stats,          "📊 Stats");
        ui.selectable_value(active_view, ComparisonView::HvHistory,      "📈 HV 履歴");
        ui.selectable_value(active_view, ComparisonView::ParetoFront,    "🔵 Pareto");
        ui.selectable_value(active_view, ComparisonView::KdeDistribution,"〜 KDE");
    });
    ui.separator();

    match active_view {
        ComparisonView::Stats          => show_stats_summary(ui, app_state),
        ComparisonView::HvHistory      => show_hv_history(ui, app_state),
        ComparisonView::ParetoFront    => show_pareto_overlay(ui, app_state),
        ComparisonView::KdeDistribution => show_kde_distribution(ui, app_state),
    }
}
```

```rust
fn show_stats_summary(ui: &mut Ui, app_state: &AppState) {
    egui::Grid::new("comparison_stats_grid")
        .striped(true)
        .show(ui, |ui| {
            // ヘッダー
            ui.strong("Study");
            ui.strong("試行数");
            ui.strong("Best 値");
            ui.strong("中央値");
            ui.strong("標準偏差");
            ui.end_row();

            for (idx, study) in app_state.comparison_studies.iter().enumerate() {
                let color = app_state.comparison_colors
                    .get(idx)
                    .copied()
                    .unwrap_or(Color32::GRAY);

                let obj_vals: Vec<f64> = study.trials.iter()
                    .filter_map(|t| t.objectives.first().copied())
                    .collect();

                let n = obj_vals.len();
                let best = obj_vals.iter().cloned().fold(f64::INFINITY, f64::min);
                let median = compute_median(&obj_vals);
                let std_dev = compute_std_dev(&obj_vals);

                ui.colored_label(color, &study.study_name);
                ui.label(n.to_string());
                ui.label(format!("{:.6}", best));
                ui.label(median.map_or("-".to_string(), |v| format!("{:.6}", v)));
                ui.label(std_dev.map_or("-".to_string(), |v| format!("{:.6}", v)));
                ui.end_row();
            }
        });
}
```

```rust
fn show_pareto_overlay(ui: &mut Ui, app_state: &AppState) {
    use egui_plot::{Plot, Points, PlotPoints};

    // 2 目的のみ対応（3 目的以上は「2D 投影表示」として説明を表示）
    let is_2d = app_state.comparison_studies.first()
        .map(|s| s.trials.first().map_or(false, |t| t.objectives.len() == 2))
        .unwrap_or(false);

    if !is_2d {
        ui.label("Pareto Front オーバーレイは 2 目的のみ対応しています");
        return;
    }

    Plot::new("pareto_overlay_plot").show(ui, |plot_ui| {
        for (idx, study) in app_state.comparison_studies.iter().enumerate() {
            let color = app_state.comparison_colors
                .get(idx)
                .copied()
                .unwrap_or(Color32::GRAY);

            // Pareto 解のみフィルタ（study に pareto_front が含まれている場合）
            let pareto_points: Vec<[f64; 2]> = study.pareto_trials.iter()
                .filter_map(|t| {
                    let objs = &t.objectives;
                    if objs.len() >= 2 {
                        Some([objs[0], objs[1]])
                    } else {
                        None
                    }
                })
                .collect();

            let plot_points = PlotPoints::new(pareto_points);
            plot_ui.points(
                Points::new(plot_points)
                    .color(color)
                    .radius(4.0)
                    .name(&study.study_name),
            );
        }
    });
}
```

### 5. KDE Distribution ビュー 🟡

**信頼性**: 🟡 *REQ-006-B-4 は KDE の実装方法が spec に詳細記載なし。ガウスカーネル近似での実装を推測*

```rust
fn show_kde_distribution(ui: &mut Ui, app_state: &AppState) {
    use egui_plot::{Plot, Line, PlotPoints};

    Plot::new("kde_plot").show(ui, |plot_ui| {
        for (idx, study) in app_state.comparison_studies.iter().enumerate() {
            let color = app_state.comparison_colors
                .get(idx)
                .copied()
                .unwrap_or(Color32::GRAY);

            let obj_vals: Vec<f64> = study.trials.iter()
                .filter_map(|t| t.objectives.first().copied())
                .collect();

            if obj_vals.is_empty() { continue; }

            let kde_curve = compute_kde(&obj_vals, 200);
            let points: PlotPoints = kde_curve.into_iter().collect();
            plot_ui.line(Line::new(points).color(color).name(&study.study_name));
        }
    });
}

/// ガウスカーネルによる KDE 曲線を生成する（resolution 点）
fn compute_kde(vals: &[f64], resolution: usize) -> Vec<[f64; 2]> {
    if vals.is_empty() { return Vec::new(); }

    let min = vals.iter().cloned().fold(f64::INFINITY, f64::min);
    let max = vals.iter().cloned().fold(f64::NEG_INFINITY, f64::max);
    let range = (max - min).max(1e-10);
    let bandwidth = 1.06 * std_dev_f64(vals) * (vals.len() as f64).powf(-0.2);

    (0..resolution).map(|i| {
        let x = min + (i as f64 / (resolution - 1) as f64) * range;
        let density: f64 = vals.iter().map(|&v| {
            let z = (x - v) / bandwidth;
            (-0.5 * z * z).exp() / (bandwidth * (2.0 * std::f64::consts::PI).sqrt())
        }).sum::<f64>() / vals.len() as f64;
        [x, density]
    }).collect()
}
```

### 6. 統計ユーティリティ関数 🔵

**信頼性**: 🔵 *Stats Summary / KDE で共通使用する標準実装*

```rust
fn compute_median(vals: &[f64]) -> Option<f64> {
    if vals.is_empty() { return None; }
    let mut sorted = vals.to_vec();
    sorted.sort_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
    let mid = sorted.len() / 2;
    Some(if sorted.len() % 2 == 0 {
        (sorted[mid - 1] + sorted[mid]) / 2.0
    } else {
        sorted[mid]
    })
}

fn compute_std_dev(vals: &[f64]) -> Option<f64> {
    if vals.len() < 2 { return None; }
    let mean = vals.iter().sum::<f64>() / vals.len() as f64;
    let variance = vals.iter().map(|&v| (v - mean).powi(2)).sum::<f64>() / vals.len() as f64;
    Some(variance.sqrt())
}

fn std_dev_f64(vals: &[f64]) -> f64 {
    compute_std_dev(vals).unwrap_or(1.0)
}
```
