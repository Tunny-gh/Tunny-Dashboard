```rust
// ツールバー（グラフ上部のボタン群）に追加
ui.horizontal(|ui| {
    // REQ-008-C: Best ライン表示トグル
    let best_label = if widget_states.optimization_history.show_best_line {
        "⭐ Best 非表示"
    } else {
        "⭐ Best 表示"
    };
    if ui.button(best_label).clicked() {
        widget_states.optimization_history.show_best_line =
            !widget_states.optimization_history.show_best_line;
    }

    // REQ-008-D: 対数スケールトグル
    let log_label = if widget_states.optimization_history.log_scale {
        "📏 線形スケール"
    } else {
        "📏 対数スケール"
    };
    if ui.button(log_label).clicked() {
        widget_states.optimization_history.log_scale =
            !widget_states.optimization_history.log_scale;
    }
});
```

### 2. Best ライン オーバーレイ 🔵

**信頼性**: 🔵 *REQ-008-C・best_trial_history データ設計より*

```rust
// egui_plot::Plot::new().show() 内部
if widget_states.optimization_history.show_best_line {
    if let Some(history) = &app_state.best_trial_history {
        let best_points: egui_plot::PlotPoints = history.iter()
            .enumerate()
            .map(|(i, &(_trial_id, best_val))| [i as f64, best_val])
            .collect();

        plot_ui.line(
            egui_plot::Line::new(best_points)
                .color(egui::Color32::GOLD)
                .width(2.0)
                .name("Best 値"),
        );
    }
}
```

### テストケース 2: Best ライン PlotPoints の生成 🔵

**信頼性**: 🔵 *REQ-008-C の折れ線データ生成より*

**Given**: `history = [(0, 1.0), (3, 0.8), (7, 0.5)]`（trial_id, best_val のタプル）
**When**: `PlotPoints` 変換処理を実行する
**Then**: X 軸が `[0.0, 1.0, 2.0]`（インデックス）、Y 軸が `[1.0, 0.8, 0.5]`

```rust
#[test]
fn test_best_line_plot_points() {
    let history = vec![(0u32, 1.0_f64), (3, 0.8), (7, 0.5)];
    let points: Vec<[f64; 2]> = history.iter()
        .enumerate()
        .map(|(i, &(_id, val))| [i as f64, val])
        .collect();
    assert_eq!(points[0], [0.0, 1.0]);
    assert_eq!(points[1], [1.0, 0.8]);
    assert_eq!(points[2], [2.0, 0.5]);
}
```
