### 2. `show_best_history_table()` 関数 🔵

**信頼性**: 🔵 *REQ-008-E/F・dataflow.md REQ-008 シーケンス図より*

```rust
/// Best 解遷移テーブル（単目的 Study 専用）
fn show_best_history_table(
    ui: &mut egui::Ui,
    app_state: &AppState,
    objective_names: &[String],
) {
    let history = match &app_state.best_trial_history {
        Some(h) if !h.is_empty() => h,
        _ => {
            ui.label("Best 解遷移データがありません");
            return;
        }
    };

    // Best 更新タイミングのみを抽出
    let best_entries = extract_best_entries(history);

    // 変数名（重要度上位 5 件 or 先頭 5 変数）
    let top_param_names: Vec<_> = app_state
        .current_study
        .as_ref()
        .map(|s| s.param_names.iter().take(5).cloned().collect())
        .unwrap_or_default();

    let obj_name = objective_names.first().map(|s| s.as_str()).unwrap_or("Objective");

    egui::ScrollArea::vertical().show(ui, |ui| {
        egui::Grid::new("best_history_grid")
            .striped(true)
            .show(ui, |ui| {
                // ヘッダー
                ui.strong("Trial ID");
                ui.strong(obj_name);
                ui.strong("Δvalue");
                for name in &top_param_names {
                    ui.strong(name);
                }
                ui.end_row();

                // データ行
                let mut prev_val: Option<f64> = None;
                for &(trial_id, best_val) in &best_entries {
                    let delta = prev_val.map(|p| best_val - p).unwrap_or(0.0);
                    prev_val = Some(best_val);

                    ui.label(format!("#{}", trial_id));
                    ui.label(format!("{:.6}", best_val));
                    ui.label(if delta == 0.0 {
                        "-".to_string()
                    } else {
                        format!("{:+.6}", delta)
                    });

                    // 変数値（trial_id に対応する試行データから取得）
                    if let Some(trial) = app_state.get_trial_by_id(trial_id) {
                        for name in &top_param_names {
                            let val = trial.params.get(name)
                                .map(|v| format!("{:.4}", v))
                                .unwrap_or_else(|| "-".to_string());
                            ui.label(val);
                        }
                    } else {
                        for _ in &top_param_names {
                            ui.label("-");
                        }
                    }
                    ui.end_row();
                }
            });
    });
}
```

### 3. `extract_best_entries` — Best 更新タイミング抽出 🔵

**信頼性**: 🔵 *REQ-008-E/F・`best_trial_history` の設計より*

```rust
/// best_trial_history から Best 値が実際に更新されたエントリのみを抽出する
/// best_trial_history は (trial_id, cumulative_best_value) の形式
fn extract_best_entries(history: &[(u32, f64)]) -> Vec<(u32, f64)> {
    let mut entries = Vec::new();
    let mut last_best: Option<f64> = None;
    for &(trial_id, best_val) in history {
        let is_new_best = last_best.map_or(true, |prev| best_val < prev);
        if is_new_best {
            entries.push((trial_id, best_val));
            last_best = Some(best_val);
        }
    }
    entries
}
```

#[test]
fn test_extract_best_entries() {
    let history = vec![(0u32, 1.0_f64), (1, 1.0), (2, 0.8), (3, 0.8), (4, 0.5)];
    let entries = extract_best_entries(&history);
    assert_eq!(entries.len(), 3);
    assert_eq!(entries[0], (0, 1.0));
    assert_eq!(entries[1], (2, 0.8));
    assert_eq!(entries[2], (4, 0.5));
}
