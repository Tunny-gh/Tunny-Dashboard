```rust
use std::path::PathBuf;
use egui::Context;

/// アーティファクトモーダルの表示状態
pub struct ArtifactModalState {
    pub open: bool,
    pub trial_id: u32,
    pub files: Vec<PathBuf>,         // 表示対象ファイル一覧
    pub selected_file_idx: usize,    // 現在選択中のファイルインデックス
    pub csv_preview: Option<Vec<Vec<String>>>, // CSV テーブルキャッシュ
    pub image_texture: Option<egui::TextureHandle>, // 画像テクスチャキャッシュ
}

impl Default for ArtifactModalState {
    fn default() -> Self {
        Self {
            open: false,
            trial_id: 0,
            files: Vec::new(),
            selected_file_idx: 0,
            csv_preview: None,
            image_texture: None,
        }
    }
}
```

### 2. `show_artifact_modal()` 関数 🔵

**信頼性**: 🔵 *REQ-007-D/E/F/G・egui Window API パターンより*

```rust
/// アーティファクトプレビューモーダルを表示する
pub fn show_artifact_modal(
    ctx: &Context,
    state: &mut ArtifactModalState,
) {
    if !state.open {
        return;
    }

    egui::Window::new(format!("Artifacts — Trial #{}", state.trial_id))
        .open(&mut state.open)
        .resizable(true)
        .min_width(400.0)
        .show(ctx, |ui| {
            // ファイル一覧タブ（複数ファイルの場合）
            if state.files.len() > 1 {
                ui.horizontal(|ui| {
                    for (i, path) in state.files.iter().enumerate() {
                        let name = path.file_name()
                            .and_then(|n| n.to_str())
                            .unwrap_or("?");
                        if ui.selectable_label(state.selected_file_idx == i, name).clicked() {
                            state.selected_file_idx = i;
                            state.csv_preview = None;
                            state.image_texture = None;
                        }
                    }
                });
                ui.separator();
            }

            // 選択ファイルのプレビュー
            if let Some(path) = state.files.get(state.selected_file_idx) {
                show_file_preview(ui, ctx, path, state);
            }
        });
}
```

```rust
fn show_file_preview(
    ui: &mut egui::Ui,
    ctx: &egui::Context,
    path: &PathBuf,
    state: &mut ArtifactModalState,
) {
    use crate::io::artifacts::ArtifactFileType;

    match ArtifactFileType::from_path(path) {
        ArtifactFileType::Image => show_image_preview(ui, ctx, path, state),
        ArtifactFileType::Csv   => show_csv_preview(ui, path, state),
        ArtifactFileType::Other => show_other_preview(ui, path),
    }
}
```

### 4. `show_image_preview` — 画像インライン表示（REQ-007-E） 🔵

**信頼性**: 🔵 *REQ-007-E・egui Image / TextureHandle API より*

```rust
fn show_image_preview(
    ui: &mut egui::Ui,
    ctx: &egui::Context,
    path: &PathBuf,
    state: &mut ArtifactModalState,
) {
    // テクスチャキャッシュが空の場合は読み込む
    if state.image_texture.is_none() {
        if let Ok(img_bytes) = std::fs::read(path) {
            if let Ok(img) = image::load_from_memory(&img_bytes) {
                let rgba = img.to_rgba8();
                let size = [rgba.width() as usize, rgba.height() as usize];
                let pixels = rgba.into_raw();
                let texture = ctx.load_texture(
                    "artifact_preview",
                    egui::ColorImage::from_rgba_unmultiplied(size, &pixels),
                    egui::TextureOptions::default(),
                );
                state.image_texture = Some(texture);
            }
        }
    }

    if let Some(texture) = &state.image_texture {
        let max_size = egui::Vec2::new(600.0, 400.0);
        ui.add(egui::Image::new(texture).fit_to_exact_size(max_size));
    } else {
        ui.label("画像を読み込めませんでした");
    }
}
```

### 5. `show_csv_preview` — CSV テーブルプレビュー（REQ-007-F） 🔵

**信頼性**: 🔵 *REQ-007-F（先頭 100 行テーブルプレビュー）より*

```rust
fn show_csv_preview(
    ui: &mut egui::Ui,
    path: &PathBuf,
    state: &mut ArtifactModalState,
) {
    // CSV キャッシュが空の場合は読み込む
    if state.csv_preview.is_none() {
        if let Ok(content) = std::fs::read_to_string(path) {
            let rows: Vec<Vec<String>> = content.lines()
                .take(100)  // REQ-007-F: 先頭 100 行
                .map(|line| line.split(',').map(|s| s.trim().to_string()).collect())
                .collect();
            state.csv_preview = Some(rows);
        }
    }

    if let Some(rows) = &state.csv_preview {
        egui::ScrollArea::both().show(ui, |ui| {
            egui::Grid::new("csv_preview_grid")
                .striped(true)
                .show(ui, |ui| {
                    for row in rows {
                        for cell in row {
                            ui.label(cell);
                        }
                        ui.end_row();
                    }
                });
        });
    } else {
        ui.label("CSV を読み込めませんでした");
    }
}
```

```rust
fn show_other_preview(ui: &mut egui::Ui, path: &PathBuf) {
    let filename = path.file_name()
        .and_then(|n| n.to_str())
        .unwrap_or("不明なファイル");

    let size_str = std::fs::metadata(path)
        .map(|m| format!("{} bytes", m.len()))
        .unwrap_or_else(|_| "サイズ不明".to_string());

    ui.label(format!("ファイル名: {}", filename));
    ui.label(format!("サイズ: {}", size_str));

    if ui.button("OS で開く").clicked() {
        let _ = open::that(path);
    }
}
```

```rust
    let rows: Vec<Vec<String>> = csv.lines()
        .take(100)
```
