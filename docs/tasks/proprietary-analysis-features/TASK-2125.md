// 各軸ヘッダーの描画ループ内に追加

for axis_name in &axis_order {
    let is_visible = *widget_states.parallel_coords.axis_visibility
        .entry(axis_name.clone())
        .or_insert(true);

    ui.horizontal(|ui| {
        // REQ-009-A: アイアイコン（表示/非表示トグル）
        let eye_icon = if is_visible { "👁" } else { "🚫" };
        if ui.button(eye_icon).on_hover_text("軸の表示/非表示を切り替え").clicked() {
            let entry = widget_states.parallel_coords.axis_visibility
                .entry(axis_name.clone())
                .or_insert(true);
            *entry = !*entry;
        }
        ui.label(axis_name);
    });
}

### 2. 軸並び替え（ドラッグ） 🟡

**信頼性**: 🟡 *REQ-009-B・egui のドラッグ API からの推測（egui に DragDrop ウィジェットが標準でないため推測)*

**実装ファイル**: `egui-app/src/ui/widgets/parallel_coords.rs`

egui の `Response.drag_delta()` を利用した軸並び替え:

```rust
// WidgetStates の parallel_coords に追加フィールドが必要（TASK-2113 で確認）
// - axis_order: Vec<String>   // 現在の軸順序
// - dragging_axis: Option<String>  // ドラッグ中の軸名
// - drag_hover_idx: Option<usize>  // ドロップ先インデックス

for (idx, axis_name) in widget_states.parallel_coords.axis_order.iter().enumerate() {
    let response = ui.label("⠿ ")  // ドラッグハンドル
        .interact(egui::Sense::drag());

    if response.drag_started() {
        widget_states.parallel_coords.dragging_axis = Some(axis_name.clone());
    }

    if response.dragged() {
        // ドラッグ中のオーバーレイ表示（省略可）
    }

    if response.drag_released() {
        // ドロップ先インデックスに移動
        if let (Some(dragged), Some(target_idx)) = (
            widget_states.parallel_coords.dragging_axis.take(),
            widget_states.parallel_coords.drag_hover_idx.take(),
        ) {
            if let Some(from_idx) = widget_states.parallel_coords.axis_order
                .iter().position(|a| a == &dragged)
            {
                let name = widget_states.parallel_coords.axis_order.remove(from_idx);
                let insert_at = target_idx.min(widget_states.parallel_coords.axis_order.len());
                widget_states.parallel_coords.axis_order.insert(insert_at, name);
            }
        }
    }
}
```

（egui 標準の DnD API に `egui::DragAndDrop` が追加されている場合はそちらを優先使用すること）

## 単体テスト要件

### テストケース 1: 軸の非表示フィルタリング 🔵

**信頼性**: 🔵 *REQ-009-A より*

**Given**: `axis_visibility = {"x1": true, "x2": false, "x3": true}`、`axis_order = ["x1", "x2", "x3"]`
**When**: `visible_axes` フィルタを実行する
**Then**: `visible_axes = ["x1", "x3"]`（x2 が除外される）

```rust
#[test]
fn test_axis_visibility_filter() {
    use std::collections::HashMap;
    let mut visibility: HashMap<String, bool> = HashMap::new();
    visibility.insert("x1".to_string(), true);
    visibility.insert("x2".to_string(), false);
    visibility.insert("x3".to_string(), true);

    let axis_order = vec!["x1".to_string(), "x2".to_string(), "x3".to_string()];
    let visible: Vec<_> = axis_order.iter()
        .filter(|name| *visibility.get(*name).unwrap_or(&true))
        .collect();

    assert_eq!(visible.len(), 2);
    assert_eq!(visible[0], "x1");
    assert_eq!(visible[1], "x3");
}
```

### テストケース 2: 軸並び替えロジック 🔵

**信頼性**: 🔵 *REQ-009-B・Vec の remove/insert パターンより*

**Given**: `axis_order = ["x1", "x2", "x3"]`、ドラッグ元 = "x1"、ドロップ先インデックス = 2
**When**: 並び替えロジックを実行する
**Then**: `axis_order = ["x2", "x3", "x1"]` （x1 が末尾に移動）

```rust
#[test]
fn test_axis_reorder() {
    let mut axis_order = vec![
        "x1".to_string(), "x2".to_string(), "x3".to_string()
    ];
    let dragged = "x1";
    let target_idx = 2;

    if let Some(from_idx) = axis_order.iter().position(|a| a == dragged) {
        let name = axis_order.remove(from_idx);
        let insert_at = target_idx.min(axis_order.len());
        axis_order.insert(insert_at, name);
    }

    assert_eq!(axis_order, vec!["x2", "x3", "x1"]);
}
```
