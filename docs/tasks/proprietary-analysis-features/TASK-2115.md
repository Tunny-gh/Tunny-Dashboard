```rust
/// フィルタ条件（filter_ranges・selected_indices・color_mode）を JSON 保存
pub fn save_filter_json(
    filter_ranges: &HashMap<String, (f64, f64)>,
    selected_indices: &[u32],
    color_mode: &str,
) {
    let snapshot = FilterSnapshot {
        filter_ranges: filter_ranges.clone(),
        selected_indices: selected_indices.to_vec(),
        color_mode: color_mode.to_string(),
    };
    if let Ok(json) = serde_json::to_string_pretty(&snapshot) {
        if let Some(path) = rfd::FileDialog::new()
            .set_file_name("filter.json")
            .add_filter("JSON", &["json"])
            .save_file()
        {
            let _ = std::fs::write(path, json);
        }
    }
}

/// フィルタ JSON を読み込んで FilterSnapshot を返す
pub fn load_filter_json() -> Option<FilterSnapshot> {
    let path = rfd::FileDialog::new()
        .add_filter("JSON", &["json"])
        .pick_file()?;
    let json = std::fs::read_to_string(path).ok()?;
    serde_json::from_str(&json).ok()
}
```

```rust
fn test_session_snapshot_backward_compat() {
    let old_json = r#"{"study_name":"test","filter_ranges":{},"selected_indices":[],"saved_at":"2025-01-01"}"#;
    let snapshot: SessionSnapshot = serde_json::from_str(old_json).unwrap();
```
