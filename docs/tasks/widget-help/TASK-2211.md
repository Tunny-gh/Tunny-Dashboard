### 1. help_modal.rs 実装 🔵

**信頼性**: 🔵 *設計文書 architecture.md・artifact_modal.rs パターンより*

**実装ファイル**: `egui-app/src/ui/help/help_modal.rs`

```rust
pub fn show_help_modal(
    ctx: &egui::Context,
    state: &mut HelpModalState,
) {
    if !state.open { return; }
    let item = match &state.item {
        Some(i) => i.clone(),
        None => return,
    };
    let content = crate::ui::help::help_content::get_help_content(&item);
    let title = format!("{} — Help", content.title);
    let mut still_open = true;
    egui::Window::new(title)
        .open(&mut still_open)
        .resizable(true)
        .min_width(500.0)
        .default_width(650.0)
        .min_height(400.0)
        .default_height(500.0)
        .show(ctx, |ui| {
            let active = state.active_tab.min(content.tabs.len().saturating_sub(1));
            ui.horizontal(|ui| {
                for (i, tab) in content.tabs.iter().enumerate() {
                    if ui.selectable_label(active == i, tab.label).clicked() {
                        state.active_tab = i;
                    }
                }
            });
            ui.separator();
            if let Some(tab) = content.tabs.get(active) {
                crate::ui::help::md_renderer::render_markdown(ui, tab.markdown);
            }
        });
    if !still_open {
        state.open = false;
        state.active_tab = 0;
    }
}
```

pub fn get_help_content(item: &PanelItem) -> HelpContent {
    match item {
        PanelItem::Chart(id) => HelpContent {
            title: id.label(),
            tabs: &[HelpTabDef {
                label: "Overview",
                markdown: "Help content coming soon.",
            }],
        },
        PanelItem::TrialTable => HelpContent {
            title: "Trial Table",
            tabs: &[HelpTabDef {
                label: "Overview",
                markdown: "Help content coming soon.",
            }],
        },
    }
}
