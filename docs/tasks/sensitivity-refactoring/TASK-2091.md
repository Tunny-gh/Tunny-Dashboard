### 1. ヘルパー関数の実装 🔵

**信頼性**: 🔵 *architecture.md・interfaces.rs より*

**実装ファイル**: `rust_core/src/sensitivity/analysis/full.rs` または `analysis/common.rs`

```rust
use crate::sensitivity::metrics::TreeMetric;
use crate::sensitivity::tree_common::prepare_training_data;
use crate::sensitivity::types::TreeImportanceResult;

fn run_tree_metric_for_all_objectives<M: TreeMetric>(
    metric: &M,
    x_matrix: &[Vec<f64>],
    objectives: &[Vec<f64>],
) -> Option<TreeImportanceResult> {
    let n_params = x_matrix.len();
    if n_params == 0 || objectives.is_empty() {
        return None;
    }

    let mut all_importances: Vec<Vec<f64>> = Vec::new();
    let mut all_r_squared: Vec<f64> = Vec::new();

    for y in objectives {
        let prepared = prepare_training_data(
            x_matrix, y, metric.max_rows(),
            metric.data_seed(), metric.split_seed()
        );

        let (importances, r2) = match prepared {
            Some(d) => metric.compute_importances(&d)
                .unwrap_or_else(|| (vec![0.0; n_params], 0.0)),
            None => (vec![0.0; n_params], 0.0),
        };

        all_importances.push(importances);
        all_r_squared.push(r2);
    }

    // transpose: [obj][param] → [param][obj]
    let importances: Vec<Vec<f64>> = (0..n_params)
        .map(|j| all_importances.iter().map(|imp| imp[j]).collect())
        .collect();

    Some(TreeImportanceResult {
        importances,
        r_squared: all_r_squared,
    })
}
```
