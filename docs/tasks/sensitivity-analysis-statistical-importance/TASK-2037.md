既存の100木ループを変更して、各木の重要度を記録する:

```rust
let mut per_tree = PerTreeImportances {
    raw: vec![vec![0.0; n_params]; n_trees],
    n_trees,
    n_params,

for t in 0..n_trees {
    for j in 0..n_params {
        let permuted_mse = eval_permuted(t, j, &predictions, x_val, y_val);
        let importance = (permuted_mse - baseline_mse).max(0.0);
        per_tree.raw[t][j] = importance;
```

### 3. 統計量計算 🔵

**信頼性**: 🔵 *REQ-STAT-031〜033・dataflow.md フロー3より*

```rust
let sum_raw: f64 = raw_means.iter().sum();
let mut p_values_raw = Vec::with_capacity(n_params);
let mut ci_lower_vec = Vec::with_capacity(n_params);
let mut ci_upper_vec = Vec::with_capacity(n_params);

let t_crit = crate::core::math::statistics::t_critical_95(99.0);

for j in 0..n_params {
    let tree_vals: Vec<f64> = (0..n_trees).map(|t| per_tree.raw[t][j]).collect();
    let mean_j = raw_means[j];
    let se_j = std_dev(&tree_vals) / (n_trees as f64).sqrt();

    let p_val = if se_j < 1e-12 {
        None  // NFR-STAT-011
    } else {
        let t_j = mean_j / se_j;
        Some(crate::core::math::statistics::t_one_sided_upper_p(t_j, 99.0))
    };
    p_values_raw.push(p_val);

    // 正規化後スケールに変換
    let norm_j = if sum_raw > 1e-12 { mean_j / sum_raw } else { 0.0 };
    let se_norm = if sum_raw > 1e-12 { se_j / sum_raw } else { 0.0 };
    ci_lower_vec.push((norm_j - t_crit * se_norm).max(0.0));  // NFR-STAT-012
    ci_upper_vec.push(norm_j + t_crit * se_norm);
}
```
