## 実装詳細

### 1. Zustand ストアの基本構造 🔵

**信頼性**: 🔵 *analysisStore.ts / clusterStore.ts パターンより*

**実装ファイル**: `frontend/src/stores/downsampleStore.ts`

```typescript
import { create } from 'zustand';
import { DOWNSAMPLE_CONFIGS, DownsampleKey } from '../types/downsampling';

interface DownsampleState { ... }
interface DownsampleActions { ... }

export const useDownsampleStore = create<DownsampleState & DownsampleActions>()(
  (set, get) => ({
    cache: {},
    isComputing: false,
    error: null,
    lastTotalCount: 0,

    reset: () => set({ cache: {}, error: null }),

    recompute: async () => {
      set({ isComputing: true });
      try {
        const wasm = await WasmLoader.getInstance();
        const results: Partial<Record<DownsampleKey, Uint32Array>> = {};
        // DOWNSAMPLE_CONFIGS の各キーに対して適切な WASM 関数を呼び出す
        for (const [key, config] of Object.entries(DOWNSAMPLE_CONFIGS)) {
          const result = callDownsampleWasm(wasm, key, config);
          results[key] = new Uint32Array(result.indices);
        }
        set({ cache: results, isComputing: false });
      } catch (e) {
        set({ error: String(e), isComputing: false });
      }
    },

    getIndices: (key) => {
      return get().cache[key] ?? new Uint32Array(); // フォールバック: 空配列
    },
  })
);
```
